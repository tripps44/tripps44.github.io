Overview

Comparative scouting analysis of the New England Patriots and Los Angeles Rams using NFL play-by-play data, built entirely in R and Quarto.

Metrics Analyzed

  • Pass/run rate splits
  • Snaps from shotgun vs under center
  • 4th down conversion tendencies
  • Goal-to-go situational analysis

Tools / Packages Used

  • nflreadr
  • tidyverse
  • ggplot2
  • flextable
  • patchwork

# creating the variables
pbp <- pbp |> 
  mutate(    dome = case_when (
    roof== "dome" ~ 1,
    roof== "closed" ~ 1,
    roof== "outdoors" ~ 0,
    roof== "open" ~ 0), 
    wind_group = case_when(
      wind >= 0 & wind <= 2 ~ "no breeze",
      dome ==1 ~ "no breeze",
      wind >= 3 & wind <= 7 ~ "light breeze",
      wind >= 8 & wind <= 14 ~ "moderate breeze",
      wind >= 15 & wind <= 25 ~ "strong breeze",
      wind > 25 ~ "strong gusts",
      TRUE ~ NA_character_), 
    wind_speed = if_else(dome==0, wind, 0), 
    fg_distance=yardline_100+17, 
    fg_group = case_when(
      fg_distance < 30  ~ "<30",
      fg_distance >= 30 & fg_distance <= 39 ~ "30-39",
      fg_distance >= 40 & fg_distance <= 44 ~ "40-44",
      fg_distance >= 45 & fg_distance <= 49 ~ "45-49",
      fg_distance >= 50 & fg_distance <= 55 ~ "50-55",
      fg_distance > 55 & fg_distance <= 60  ~ "55-60",
      fg_distance > 60   ~ "60+",
      TRUE ~ NA_character_), 
    fg_50_plus = if_else(fg_distance >49, 1, 0),  
    fg_long = if_else(fg_distance >54, 1, 0),  
    fg_made = case_when(
      field_goal_result == "made" ~ 1,
      field_goal_result == "missed" ~ 0,
      field_goal_result == "blocked" ~ 0,
      TRUE ~ NA_real_  ), 
    field_position_group = case_when(
      yardline_100 <=10 ~ "inside 10",
      yardline_100 >10 & yardline_100 <= 20 ~ "red zone",
      yardline_100 > 20 & yardline_100 <= 45 ~ "driving zone",
      yardline_100 > 45 & yardline_100 <= 55 ~ "midfield",
      yardline_100 > 55 & yardline_100 <= 80 ~ "opp zone",
      yardline_100 > 80 & yardline_100 <= 95 ~ "opp red zone",
      yardline_100 > 95 & yardline_100 <= 99 ~ "inside opp 5",
      TRUE ~ NA_character_), 
    drive_points = case_when(
      fixed_drive_result == "Touchdown" ~ 7,
      fixed_drive_result == "Field goal" ~ 3,
      fixed_drive_result == "Safety" ~ -2,
      fixed_drive_result == "Opp touchdown" ~ -7,
      TRUE ~ 0), 
    to_go = case_when(
      ydstogo == 1 ~ 1,
      ydstogo %in% 2:3 ~ 2,
      ydstogo %in% 4:6 ~ 3,
      ydstogo %in% 7:9 ~ 4,
      ydstogo %in% 10:14 ~ 5,
      ydstogo >= 15 ~ 6,
      TRUE ~ NA_real_ ),
    to_go_f = factor(to_go, 
                     levels = 1:6, 
                     labels = c("1 yard", "2-3 yards", "4-6 yards", "7-9 yards", "10-14 yards", "15+ yards")))

# specific dataframes for first half only

# All Team's Deep Dive on Passing Tendencies in First Half
rate_all <- pbp |> 
  filter(
    down %in% 1:3,
    game_seconds_remaining > 1799,
    pass_attempt %in% c(0,1) | rush_attempt %in% c(0,1),
    goal_to_go==0
  ) |> 
  group_by(down, to_go_f) |> 
  summarise(
    pass_percent = mean(pass_attempt, na.rm = TRUE),
    pass_att = sum(pass_attempt, na.rm = TRUE),
    plays = n(),
    .groups = "drop"
  )

rate_all.plot <- ggplot(rate_all, aes(x = to_go_f, y = factor(down), fill = pass_percent)) +
  geom_tile(color = "white") +
  geom_text(
    aes(label = paste0(
      scales::percent(pass_percent, accuracy = 1), "\n(",
      plays, ")")),
    color = "black",
    lineheight = 0.9,
    size = 4) +
  scale_fill_gradient(
    low = "antiquewhite",
    high = "#46b478",
    labels = scales::percent_format(accuracy = 1)) +
  labs(x = "Yards to Go",
       y = "Down",
       fill = "Pass %",
       title = "NFL Probability of Passing by Down and Distance (2025)") +
  theme_minimal(base_size = 14) +
  theme(axis.title.y = element_text(angle = 0, vjust = 0.5),
        panel.grid = element_blank())

Compared to the Patriots, LA seems to pass the ball a lot more than average (from shotgun) and on 2nd down and 2 to 9 yards to go (under center). This is something opposing teams can take into account.

## Overview
Comparative scouting analysis of the New England Patriots and 
Los Angeles Rams using NFL play-by-play data, built entirely in R and Quarto.

## Metrics Analyzed
- Pass/run rate splits
- Snaps from shotgun vs under center
- 4th down conversion tendencies
- Goal-to-go situational analysis

## Tools / Packages Used
- nflreadr
- tidyverse
- ggplot2
- flextable
- patchwork

```{r}
# creating the variables
pbp <- pbp |> 
  mutate(    dome = case_when (
    roof== "dome" ~ 1,
    roof== "closed" ~ 1,
    roof== "outdoors" ~ 0,
    roof== "open" ~ 0), 
    wind_group = case_when(
      wind >= 0 & wind <= 2 ~ "no breeze",
      dome ==1 ~ "no breeze",
      wind >= 3 & wind <= 7 ~ "light breeze",
      wind >= 8 & wind <= 14 ~ "moderate breeze",
      wind >= 15 & wind <= 25 ~ "strong breeze",
      wind > 25 ~ "strong gusts",
      TRUE ~ NA_character_), 
    wind_speed = if_else(dome==0, wind, 0), 
    fg_distance=yardline_100+17, 
    fg_group = case_when(
      fg_distance < 30  ~ "<30",
      fg_distance >= 30 & fg_distance <= 39 ~ "30-39",
      fg_distance >= 40 & fg_distance <= 44 ~ "40-44",
      fg_distance >= 45 & fg_distance <= 49 ~ "45-49",
      fg_distance >= 50 & fg_distance <= 55 ~ "50-55",
      fg_distance > 55 & fg_distance <= 60  ~ "55-60",
      fg_distance > 60   ~ "60+",
      TRUE ~ NA_character_), 
    fg_50_plus = if_else(fg_distance >49, 1, 0),  
    fg_long = if_else(fg_distance >54, 1, 0),  
    fg_made = case_when(
      field_goal_result == "made" ~ 1,
      field_goal_result == "missed" ~ 0,
      field_goal_result == "blocked" ~ 0,
      TRUE ~ NA_real_  ), 
    field_position_group = case_when(
      yardline_100 <=10 ~ "inside 10",
      yardline_100 >10 & yardline_100 <= 20 ~ "red zone",
      yardline_100 > 20 & yardline_100 <= 45 ~ "driving zone",
      yardline_100 > 45 & yardline_100 <= 55 ~ "midfield",
      yardline_100 > 55 & yardline_100 <= 80 ~ "opp zone",
      yardline_100 > 80 & yardline_100 <= 95 ~ "opp red zone",
      yardline_100 > 95 & yardline_100 <= 99 ~ "inside opp 5",
      TRUE ~ NA_character_), 
    drive_points = case_when(
      fixed_drive_result == "Touchdown" ~ 7,
      fixed_drive_result == "Field goal" ~ 3,
      fixed_drive_result == "Safety" ~ -2,
      fixed_drive_result == "Opp touchdown" ~ -7,
      TRUE ~ 0), 
    to_go = case_when(
      ydstogo == 1 ~ 1,
      ydstogo %in% 2:3 ~ 2,
      ydstogo %in% 4:6 ~ 3,
      ydstogo %in% 7:9 ~ 4,
      ydstogo %in% 10:14 ~ 5,
      ydstogo >= 15 ~ 6,
      TRUE ~ NA_real_ ),
    to_go_f = factor(to_go, 
                     levels = 1:6, 
                     labels = c("1 yard", "2-3 yards", "4-6 yards", "7-9 yards", "10-14 yards", "15+ yards")))
```



```{r}
# specific dataframes for first half only

# All Team's Deep Dive on Passing Tendencies in First Half
rate_all <- pbp |> 
  filter(
    down %in% 1:3,
    game_seconds_remaining > 1799,
    pass_attempt %in% c(0,1) | rush_attempt %in% c(0,1),
    goal_to_go==0
  ) |> 
  group_by(down, to_go_f) |> 
  summarise(
    pass_percent = mean(pass_attempt, na.rm = TRUE),
    pass_att = sum(pass_attempt, na.rm = TRUE),
    plays = n(),
    .groups = "drop"
  )

rate_all.plot <- ggplot(rate_all, aes(x = to_go_f, y = factor(down), fill = pass_percent)) +
  geom_tile(color = "white") +
  geom_text(
    aes(label = paste0(
      scales::percent(pass_percent, accuracy = 1), "\n(",
      plays, ")")),
    color = "black",
    lineheight = 0.9,
    size = 4) +
  scale_fill_gradient(
    low = "antiquewhite",
    high = "#46b478",
    labels = scales::percent_format(accuracy = 1)) +
  labs(x = "Yards to Go",
       y = "Down",
       fill = "Pass %",
       title = "NFL Probability of Passing by Down and Distance (2025)") +
  theme_minimal(base_size = 14) +
  theme(axis.title.y = element_text(angle = 0, vjust = 0.5),
        panel.grid = element_blank())
```

The heatmaps show that both the Patriots and Rams use the shotgun formation more 
than under center overall.

formation.

Compared to the Patriots, LA seems to pass the ball a lot more than average (from shotgun) 
and on 2nd down and 2 to 9 yards to go (under center). 
This is something opposing teams can take into account.
