def get_pre_shot_passes(passes_df, shots_df):
    """Get passes made within 15 seconds of a shot"""

    pre_shot = [] # empty list to collect results

    for _, shot in shots_df.iterrows():
        #looping through every shot one at a time

        shot_time = shot['minute'] * 60 + shot['second']
        # converting shot time to total seconds

        window = passes_df[
            (passes_df['minute'] * 60 + passes_df['second'])
            .between(shot_time - 15, shot_time)
        
        #convert every pass time to total seconds
        # then find passes that happened within the 15 second window BEFORE the shot
        ]
        pre_shot.append(window)
        # add to results list

    return pd.concat(pre_shot).drop_duplicates() if pre_shot else pd.DataFrame()
        # combine all results
        # remove duplicates
        # return empty dataframe if nothing is found 

arg_pre_shot = get_pre_shot_passes(arg_passes, arg_shots)

Why Track passes 15 seconds before a shot?

This type of tracking reveals the precise off ball movements, spacing and defensive fractures that create scoring opportunities, rather than just analyzing the final action. If the coach only looks at shots, then they’re reacting only to the outcome. Looking at passes 15 seconds before a shot can give an idea of spatial patterns before looking at the shot map

🟦 Argentina Takeaways

  1. The high volume of passes show that Argentina dominated possession
  2. Short and quick arrows are clustered together meaning a lot of tight passing was carried out
  3. Argentina controlled the ball and recycled possession heavily in the opponent’s half creating a sustained attacking pressure

🟩 Saudi Takeways

Saudi Arabia sat deeper, played fewer passes and relied on direct balls when they played forward

def draw_passing_network(ax, passes_df, team, color, top5):
    """Passing network for top 5 players"""
    
    # passes between top 5 players only
    complete = passes_df[
        passes_df['outcome_name'].isna() &
        passes_df['pass_recipient_name'].notna() &
        passes_df['player_name'].isin(top5) &
        passes_df['pass_recipient_name'].isin(top5)
    ]

    # pass pairs
    pairs = complete.groupby(
        ['player_name', 'pass_recipient_name']
    ).size().reset_index(name='pass_count')

    # average positions
    avg_pos = complete.groupby('player_name').agg(
        avg_x=('x', 'mean'),
        avg_y=('y', 'mean'),
        total=('x', 'count')
    ).reset_index()

    pitch = Pitch(
        pitch_type='statsbomb',
        pitch_color='#fafafa',
        line_color='#cccccc'
    )
    pitch.draw(ax=ax)

    # draw lines
    for _, row in pairs.iterrows():
        p1 = avg_pos[avg_pos['player_name'] == row['player_name']]
        p2 = avg_pos[avg_pos['player_name'] == row['pass_recipient_name']]
        if not p1.empty and not p2.empty:
            pitch.lines(
                p1.avg_x.values[0], p1.avg_y.values[0],
                p2.avg_x.values[0], p2.avg_y.values[0],
                ax=ax,
                lw=row['pass_count'] * 0.4,
                color=color,
                alpha=0.5,
                zorder=1
            )

    # draw nodes
    pitch.scatter(
        avg_pos.avg_x, avg_pos.avg_y,
        s=avg_pos.total * 8,
        c=color,
        edgecolors='white',
        linewidth=1,
        ax=ax,
        zorder=3
    )

    # Labels — last name only
    texts = []
    for _, row in avg_pos.iterrows():
        last_name = row['player_name'].split()[-1]
        txt = ax.text(
            row['avg_x'], row['avg_y'],
            last_name,
            color='black',
            fontsize=10,
            fontweight='bold',
            ha='center',
            va='bottom',
            zorder=5
        )
        texts.append(txt)

    # adjustText: avoids label overlap
    # similar to ggrepel 
    adjust_text(
        texts,
        ax=ax,
        expand=(1.2, 1.4),
        arrowprops=dict(
            arrowstyle='-',
            color='#999999',
            lw=0.5
        )
    )

    ax.set_title(team, color='black',
                 fontsize=12, pad=12)
    return ax

Argentina’s top 5 passers were almost entirely confined to their own defensive third, a striking visual that tells the story of Saudi Arabia’s relentless high press. Otamendi and Romero, both center backs, were forced into heavy ball circulation at the back, with Paredes and De Paul dropping deep to offer passing options rather than driving forward. The tightly clustered network suggests Argentina struggled to progress the ball through midfield and build attacks from deep (an unusual sight for a team of their quality)

Saudi Arabia’s network tells the opposite story. Their top 5 passers operated across a wider area of the pitch, with Owais distributing from deep while Al-Malki, Kanoo and Hamid formed an active midfield triangle in the center of the pitch

        if not p1.empty and not p2.empty:
            pitch.lines(
                p1.avg_x.values[0], p1.avg_y.values[0],
                p2.avg_x.values[0], p2.avg_y.values[0],
                ax=ax,
                lw=row['pass_count'] * 0.4,
                color=color,
                alpha=0.5,
                zorder=1
            )

    # draw nodes
    pitch.scatter(
        avg_pos.avg_x, avg_pos.avg_y,
        s=avg_pos.total * 8,
        c=color,
        edgecolors='white',
        linewidth=1,
        ax=ax,
        zorder=3
    )

    # Labels — last name only
    texts = []
    for _, row in avg_pos.iterrows():
        last_name = row['player_name'].split()[-1]
        txt = ax.text(
            row['avg_x'], row['avg_y'],
            last_name,
            color='black',
            fontsize=10,
            fontweight='bold',
            ha='center',
            va='bottom',
            zorder=5
        )
        texts.append(txt)

    # adjustText: avoids label overlap
    # similar to ggrepel 
    adjust_text(
        texts,
        ax=ax,
        expand=(1.2, 1.4),
        arrowprops=dict(
            arrowstyle='-',
            color='#999999',
            lw=0.5
        )
    )

    ax.set_title(team, color='black',
                 fontsize=12, pad=12)
    return ax

Argentina's top 5 passers were almost entirely confined to their 
own defensive third, a striking visual that tells the story of 
Saudi Arabia's relentless high press. Otamendi and Romero, both 
center backs, were forced into heavy ball circulation at the back, 
with Paredes and De Paul dropping deep to offer passing options 
rather than driving forward. The tightly clustered network suggests 
Argentina struggled to progress the ball through midfield and build 
attacks from deep (an unusual sight for a team of their quality)

Saudi Arabia's network tells the opposite story. Their top 5 passers 
operated across a wider area of the pitch, with Owais distributing 
from deep while Al-Malki, Kanoo and Hamid formed an active midfield 
triangle in the center of the pitch
